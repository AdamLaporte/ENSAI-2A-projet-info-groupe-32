skinparam packageStyle rectangle

actor "Utilisateur Inscrit" as User
actor "Utilisateur Externe" as Public

rectangle "Système QR Code Tracking" {

  ' --- Authentification ---
  usecase "S'inscrire / Se connecter" as Auth

  ' --- Gestion des QR Codes ---
  usecase "Créer un QR Code" as Create
  usecase "Modifier un QR Code" as Update
  usecase "Supprimer un QR Code" as Delete
  usecase "Lister mes QR Codes" as List

  ' --- Fonctionnalités spécifiques ---
  usecase "Personnaliser (Couleur/Logo)" as Pimp
  usecase "Définir le type (Suivi/Non-suivi)" as TypeChoice

  ' --- Statistiques ---
  usecase "Consulter les statistiques" as Stats

  ' --- Action Publique ---
  usecase "Scanner le QR Code" as Scan
  usecase "Redirection & Log automatique" as AutoLog
}

' Relations Acteur -> Cas d'utilisation
User --> Create
User --> Update
User --> Delete
User --> List
User --> Stats

Public --> Scan

' Relations entre Cas d'utilisation (Include / Extend)

' Pour tout faire, il faut être connecté
Create ..> Auth : <<include>>
Update ..> Auth : <<include>>
Delete ..> Auth : <<include>>
List ..> Auth : <<include>>
Stats ..> Auth : <<include>>

' La personnalisation étend la création et la modification
Pimp .> Create : <<extend>>
Pimp .> Update : <<extend>>

' Le choix du type se fait à la création
TypeChoice .> Create : <<include>>

' Le scan déclenche automatiquement le log (c'est le système qui le fait)
Scan ..> AutoLog : <<include>>

@enduml
